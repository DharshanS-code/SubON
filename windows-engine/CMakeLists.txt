cmake_minimum_required(VERSION 3.20)
project(SubOnWindowsEngine VERSION 1.0.0 LANGUAGES CXX)
option(SUBON_BUILD_TESTS "Build DSP, format and native integration regression tests" ON)
option(SUBON_BUILD_DESKTOP_APP "Build Windows desktop prototype" ON)
option(SUBON_BUILD_VIRTUAL_AUDIO_ROUTER "Build experimental WASAPI virtual audio router" ON)
option(SUBON_BUILD_WINDOWS_APO_ADAPTER "Compile the Windows SDK APO connection adapter" ON)
option(SUBON_BUILD_NATIVE_APO "Build the native Windows APO COM DLL" ON)
add_library(subon_dsp STATIC src/SubOnDsp.cpp)
target_include_directories(subon_dsp PUBLIC include)
target_compile_features(subon_dsp PUBLIC cxx_std_17)
set_target_properties(subon_dsp PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(subon_dsp PRIVATE /W4 /permissive-)
else()
 target_compile_options(subon_dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(SUBON_BUILD_TESTS)
 enable_testing()
 add_executable(subon_dsp_smoke_test tests/smoke_test.cpp)
 target_link_libraries(subon_dsp_smoke_test PRIVATE subon_dsp)
 add_test(NAME subon_dsp_smoke_test COMMAND subon_dsp_smoke_test)
 add_executable(subon_interleaved_float_test tests/interleaved_float_test.cpp)
 target_link_libraries(subon_interleaved_float_test PRIVATE subon_dsp)
 add_test(NAME subon_interleaved_float_test COMMAND subon_interleaved_float_test)
 add_executable(subon_apo_dsp_bridge_test tests/apo_dsp_bridge_test.cpp)
 target_link_libraries(subon_apo_dsp_bridge_test PRIVATE subon_dsp)
 add_test(NAME subon_apo_dsp_bridge_test COMMAND subon_apo_dsp_bridge_test)
 add_executable(subon_apo_format_test tests/apo_format_test.cpp)
 target_include_directories(subon_apo_format_test PRIVATE include)
 target_compile_features(subon_apo_format_test PRIVATE cxx_std_17)
 add_test(NAME subon_apo_format_test COMMAND subon_apo_format_test)
 add_executable(subon_apo_realtime_processor_test tests/apo_realtime_processor_test.cpp)
 target_link_libraries(subon_apo_realtime_processor_test PRIVATE subon_dsp)
 add_test(NAME subon_apo_realtime_processor_test COMMAND subon_apo_realtime_processor_test)
 add_executable(subon_wav_file_test tests/WavFileTest.cpp)
 target_compile_features(subon_wav_file_test PRIVATE cxx_std_17)
 add_test(NAME subon_wav_file_test COMMAND subon_wav_file_test)
 if(WIN32 AND SUBON_BUILD_NATIVE_APO)
  add_executable(subon_windows_apo_test tests/windows_apo_test.cpp src/SubOnNativeApo.cpp)
  target_link_libraries(subon_windows_apo_test PRIVATE subon_dsp ole32 uuid)
  target_compile_definitions(subon_windows_apo_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  add_test(NAME subon_windows_apo_test COMMAND subon_windows_apo_test)
 endif()
endif()
if(WIN32 AND SUBON_BUILD_WINDOWS_APO_ADAPTER)
 add_library(subon_windows_apo_adapter OBJECT src/SubOnWindowsApoConnection.cpp)
 target_include_directories(subon_windows_apo_adapter PRIVATE include)
 target_link_libraries(subon_windows_apo_adapter PRIVATE subon_dsp)
 target_compile_features(subon_windows_apo_adapter PRIVATE cxx_std_17)
 target_compile_definitions(subon_windows_apo_adapter PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 if(MSVC)
  target_compile_options(subon_windows_apo_adapter PRIVATE /W4 /permissive-)
 endif()
endif()
if(WIN32 AND SUBON_BUILD_NATIVE_APO)
 add_library(SubONApo SHARED src/SubOnNativeApo.cpp src/SubOnWindowsApoConnection.cpp src/SubONApo.def)
 target_include_directories(SubONApo PRIVATE include)
 target_link_libraries(SubONApo PRIVATE subon_dsp ole32 uuid)
 target_compile_features(SubONApo PRIVATE cxx_std_17)
 target_compile_definitions(SubONApo PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
 set_target_properties(SubONApo PROPERTIES OUTPUT_NAME "SubONApo")
 if(MSVC)
  target_compile_options(SubONApo PRIVATE /W4 /permissive-)
 endif()
endif()
if(WIN32 AND SUBON_BUILD_DESKTOP_APP)
 add_executable(SubONWindows WIN32 src/WindowsApp.cpp)
 target_link_libraries(SubONWindows PRIVATE subon_dsp comctl32 comdlg32 winmm uxtheme)
 target_compile_features(SubONWindows PRIVATE cxx_std_17)
 target_compile_definitions(SubONWindows PRIVATE UNICODE _UNICODE)
 if(MINGW)
  target_link_options(SubONWindows PRIVATE -municode)
 endif()
 if(MSVC)
  target_compile_options(SubONWindows PRIVATE /W4 /permissive-)
 endif()
endif()
if(WIN32 AND SUBON_BUILD_VIRTUAL_AUDIO_ROUTER)
 add_executable(SubONVirtualRouter src/VirtualAudioRouter.cpp)
 target_link_libraries(SubONVirtualRouter PRIVATE subon_dsp ole32 uuid propsys)
 target_compile_features(SubONVirtualRouter PRIVATE cxx_std_17)
 target_compile_definitions(SubONVirtualRouter PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 if(MINGW)
  target_link_options(SubONVirtualRouter PRIVATE -municode)
 endif()
 if(MSVC)
  target_compile_options(SubONVirtualRouter PRIVATE /W4 /permissive-)
 endif()
endif()
